> class Solution:
> 	def getRow(self, rowIndex: int) -> List[int]:
> 		# Initialise empty row.
> 		pRow = [1] * (rowIndex + 1)
>
> 		# Dynamically find out final row.
> 		for i in range(1, rowIndex):
> 			# Work backwards so new values don't effect future results.
> 			for j in range(i, 0, -1):
> 				pRow[j] = pRow[j - 1] + pRow[j]
>
> 		return pRow
> ```
